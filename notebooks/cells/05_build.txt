tr = np.asarray(train[CHAN], dtype=np.float32)

if tr.ndim == 1:
    tr = tr[:, None]
if te.ndim == 1:
    te = te[:, None]
if tr.ndim != 2 or te.ndim != 2 or tr.shape[1] != te.shape[1]:
    raise ValueError(f"incompatible channel shapes: train={tr.shape}, test={te.shape}")

# Fit normalization on training telemetry only; never use test statistics.
mu = tr.mean(axis=0, keepdims=True)
sd = tr.std(axis=0, keepdims=True) + 1e-6


def standardize_finite(values, name):
    scaled = (values - mu) / sd
    invalid = ~np.isfinite(scaled)
    count = int(invalid.sum())
    print(f"{name}: non-finite values repaired: {count}")
    # Explicit values are essential: np.nan_to_num's default +/-inf handling
    # produces the largest representable float and can create billion-scale
    # residual z-scores and false fault alerts.
    return np.nan_to_num(scaled, nan=0.0, posinf=0.0, neginf=0.0)
